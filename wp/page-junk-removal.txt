(function () {
  [].forEach.call(document.querySelectorAll('[data-r9tabs]'), function (w) {
    var tabs = [].slice.call(w.querySelectorAll('[role="tab"]'));
    var panels = tabs.map(function (t) { return document.getElementById(t.getAttribute('aria-controls')); });
    function pick(i, focus) {
      tabs.forEach(function (t, j) { var on = i === j; t.setAttribute('aria-selected', on ? 'true' : 'false'); t.tabIndex = on ? 0 : -1; panels[j].hidden = !on; });
      if (focus) tabs[i].focus();
    }
    tabs.forEach(function (t, i) {
      t.addEventListener('click', function () { pick(i); });
      t.addEventListener('keydown', function (e) {
        var n = tabs.length, k = e.key;
        if (k === 'ArrowRight' || k === 'ArrowDown') { e.preventDefault(); pick((i + 1) % n, true); }
        else if (k === 'ArrowLeft' || k === 'ArrowUp') { e.preventDefault(); pick((i - 1 + n) % n, true); }
        else if (k === 'Home') { e.preventDefault(); pick(0, true); }
        else if (k === 'End') { e.preventDefault(); pick(n - 1, true); }
      });
    });
  });
})();






/* Conversion tracking, prepared. On jeffsjunk.ca (not /staging) it sends GA4 events through gtag or the GTM dataLayer.
   Anywhere else (this mockup, the GitHub preview, staging) it sends nothing: it only keeps a list in window.jjTrackLog. */
(function(){
  var live = /(^|\.)jeffsjunk\.ca$/.test(location.hostname) && location.pathname.indexOf('/staging') !== 0;
  var device = (window.matchMedia && matchMedia('(hover: hover) and (pointer: fine)').matches) ? 'desktop' : 'mobile';
  window.jjTrackLog = window.jjTrackLog || [];
  function send(name, p){
    p = p || {}; p.device_type = device; p.page_name = 'junk_removal';
    window.jjTrackLog.push([name, p]);
    if (!live) { if (window.console && console.debug) console.debug('[tracking ready, not sent]', name, p); return; }
    if (typeof window.gtag === 'function') window.gtag('event', name, p);
    else { window.dataLayer = window.dataLayer || []; var o = {event: name}; for (var k in p) o[k] = p[k]; window.dataLayer.push(o); }
  }
  function area(a){
    var t = a.closest('[data-track]'); if (t) return t.getAttribute('data-track');
    if (a.closest('.jj-call')) return 'sticky_bar';
    if (a.closest('.site-head, .mnav')) return 'header';
    if (a.closest('footer')) return 'footer';
    var s = a.closest('section[id]'); return s ? s.id : 'page';
  }
  document.addEventListener('click', function(e){
    var a = e.target.closest && e.target.closest('a[href]'); if (!a) return;
    var h = a.getAttribute('href'), where = area(a), p = {link_area: where};
    var est = window.jjEstState;
    if (where === 'estimator' && est) { p.est_level = est.k; p.est_price = est.price; }
    if (h.indexOf('tel:') === 0) { send('call_click', p); }
    else if (h.indexOf('sms:') === 0) { send('text_click', p); }
    else if (h.indexOf('mailto:') === 0) { send('email_click', p); }
    else if (/#(estimate|pricing|request)$/.test(h)) { send('quote_start', p); return; }
    else return;
    if (where === 'estimator') send('estimator_contact', p);
  }, true);
  // the photo form: counted when it passes the browser's checks (the form's own script then sends it, or on the mockup shows "nothing was sent")
  document.addEventListener('submit', function(e){
    var f = e.target; if (!f.matches || !f.matches('#request form.req') || !f.checkValidity()) return;
    var s = window.jjEstState, p = {link_area: 'photo_form', photos: (f._jjPhotos || []).length};
    if (s) { p.est_level = s.k; p.est_price = s.price; }
    send('quote_request', p);
  }, true);
  var range = document.getElementById('fill'), used = false;
  function estUse(){ var s = window.jjEstState; if (s) send('estimator_use', {est_level: s.k, est_price: s.price, first: !used}); used = true; }
  if (range) range.addEventListener('change', estUse);
  document.querySelectorAll('.ticks button').forEach(function(b){ b.addEventListener('click', function(){ setTimeout(estUse, 0); }); });
})();
