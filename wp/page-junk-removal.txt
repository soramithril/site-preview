(function () {
  [].forEach.call(document.querySelectorAll('[data-r9tabs]'), function (w) {
    var tabs = [].slice.call(w.querySelectorAll('[role="tab"]'));
    var panels = tabs.map(function (t) { return document.getElementById(t.getAttribute('aria-controls')); });
    function pick(i, focus) {
      tabs.forEach(function (t, j) { var on = i === j; t.setAttribute('aria-selected', on ? 'true' : 'false'); t.tabIndex = on ? 0 : -1; panels[j].hidden = !on; });
      if (focus) tabs[i].focus();
    }
    tabs.forEach(function (t, i) {
      t.addEventListener('click', function () { pick(i); });
      t.addEventListener('keydown', function (e) {
        var n = tabs.length, k = e.key;
        if (k === 'ArrowRight' || k === 'ArrowDown') { e.preventDefault(); pick((i + 1) % n, true); }
        else if (k === 'ArrowLeft' || k === 'ArrowUp') { e.preventDefault(); pick((i - 1 + n) % n, true); }
        else if (k === 'Home') { e.preventDefault(); pick(0, true); }
        else if (k === 'End') { e.preventDefault(); pick(n - 1, true); }
      });
    });
  });
})();


(function(){
  document.querySelectorAll('[data-quiz]').forEach(function(q){
    var steps = q.querySelectorAll('[data-step-id]'), hist = [];
    function light(lit){
      q.querySelectorAll('[data-bin]').forEach(function(f){
        if (lit === null) { f.classList.remove('lit', 'dim'); return; }
        var l = lit.split(' ').indexOf(f.dataset.bin) >= 0; f.classList.toggle('lit', l); f.classList.toggle('dim', !l); });
    }
    function go(id, push){
      var cur = q.querySelector('[data-step-id]:not([hidden])');
      if (push && cur) hist.push(cur.dataset.stepId);
      steps.forEach(function(s){ s.hidden = s.dataset.stepId !== id; });
      var now = q.querySelector('[data-step-id="' + id + '"]');
      q.classList.toggle('done', id.charAt(0) === 'r');
      if (id === 's0') { hist = []; light(null); } else if (id === 's1') light(null);
      else if (now && now.dataset.lit !== undefined) light(now.dataset.lit);
      var f = now && now.querySelector('button, a'); if (f && push) f.focus({preventScroll: true});
    }
    q.addEventListener('click', function(e){
      var b = e.target.closest('[data-go]'), k = e.target.closest('[data-back]');
      if (b) { if (b.dataset.lit !== undefined) light(b.dataset.lit); go(b.dataset.go, b.dataset.go !== 's0'); }
      else if (k) { go(hist.pop() || 's0', false); }
    });
  });
})();
