    });

      if (q === a || (q.length >= 3 && a.indexOf(q) === 0)) { var t = byName(ALIASES[a]); if (t && out.indexOf(t) < 0) out.unshift(t); }
    });
    return out;
  }

  }).join('');

  function hl(name, q){

      var t = Math.min(1, (ts - t0) / dur), e = 1 - Math.pow(1 - t, 4);

    rv.forEach(function(el){ el.classList.add('in'); });
  }
})();
